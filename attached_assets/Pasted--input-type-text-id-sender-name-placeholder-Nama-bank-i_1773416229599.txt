<script>
   

    const QRIS_BASE = "00020101021126610014COM.GO-JEK.WWW01189360091439663050810210G9663050810303UMI51440014ID.CO.QRIS.WWW0215ID10254671365660303UMI5204549953033605802ID5917ALTOMEDIA, Grosir6008KARAWANG61054136162070703A016304D21A";
    let qrisTimer;

    function setNominal(val) {
        document.getElementById('amount-input').value = val;
        document.querySelectorAll('.btn-nominal').forEach(b => b.classList.remove('active'));
        event.currentTarget.classList.add('active');
    }

    function crc16(data) {
        let crc = 0xFFFF;
        for (let i = 0; i < data.length; i++) {
            crc ^= data.charCodeAt(i) << 8;
            for (let j = 0; j < 8; j++) {
                if ((crc & 0x8000) !== 0) crc = (crc << 1) ^ 0x1021;
                else crc <<= 1;
            }
        }
        return (crc & 0xFFFF).toString(16).toUpperCase().padStart(4, '0');
    }

    function generateQR() {
        const amount = document.getElementById('amount-input').value;
        const name = document.getElementById('sender-name').value || "Supporter";
        const message = document.getElementById('sender-message').value || "";
        
        if (!amount || amount < 1000) return alert("Minimal topup Rp 1.0000");

        document.getElementById('cheer-sound').play();
        const msgVoice = new SpeechSynthesisUtterance("silahkan simpan atau screenshoot kris untuk memberikan dukungan");
        msgVoice.lang = 'id-ID';
        window.speechSynthesis.speak(msgVoice);
        confetti({ particleCount: 150, spread: 70, origin: { y: 0.6 }, colors: ['#c5a059', '#fff'] });

        let qrisTanpaCRC = QRIS_BASE.split("6304")[0];
        let tagNominal = "54" + amount.length.toString().padStart(2, '0') + amount;
        let dataSiapCRC = qrisTanpaCRC + tagNominal + "6304";
        let fullQRIS = dataSiapCRC + crc16(dataSiapCRC);

        document.getElementById('qrcode').innerHTML = "";
        new QRCode(document.getElementById('qrcode'), { text: fullQRIS, width: 200, height: 200, correctLevel: QRCode.CorrectLevel.M });

        document.getElementById('total-display').innerText = "Rp " + parseInt(amount).toLocaleString('id-ID');
        document.getElementById('tr-id').innerText = "TRX-ID: " + Math.floor(Date.now()/1000);
        document.getElementById('qr-container').style.display = 'block';

        // Firebase Push Logic
        if (window.saveDonation) {
            window.saveDonation(name, amount, message);
        }

        let time = 180;
        clearInterval(qrisTimer);
        qrisTimer = setInterval(() => {
            let m = Math.floor(time/60), s = time%60;
            document.getElementById('pay-timer').innerText = `Selesaikan dalam ${m}:${s<10?'0':''}${s}`;
            if(time-- <= 0) { clearInterval(qrisTimer); document.getElementById('qr-container').style.opacity="0.5"; alert("Waktu habis."); }
        }, 1000);

        window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
    }

    function downloadQR() {
        html2canvas(document.getElementById('capture-area')).then(canvas => {
            const link = document.createElement('a');
            link.download = `QRIS-SIDHANIE-${Date.now()}.png`;
            link.href = canvas.toDataURL();
            link.click();
        });
    }

    const canvas = document.getElementById('stars');
    const ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth; canvas.height = window.innerHeight;
    const stars = Array(30).fill().map(() => ({ x: Math.random()*canvas.width, y: Math.random()*canvas.height, s: Math.random()*0.4 }));
    function draw() {
        ctx.clearRect(0,0,canvas.width,canvas.height); ctx.fillStyle = "#c5a059";
        stars.forEach(s => { ctx.beginPath(); ctx.arc(s.x, s.y, 1.2, 0, Math.PI*2); ctx.fill(); s.y -= s.s; if(s.y<0) s.y=canvas.height; });
        requestAnimationFrame(draw);
    }
    draw();
</script>

<script type="module">
   
    window.saveDonation = async function(name, amount, message) {
        try {
            await addDoc(collection(db, "donations"), {
                name: name,
                amount: parseInt(amount),
                message: message,
                timestamp: new Date()
            });
        } catch (e) { console.error("Error: ", e); }
    }

    const q = query(collection(db, "donations"), orderBy("timestamp", "desc"), limit(10));
    onSnapshot(q, (snapshot) => {
        const list = document.getElementById('donation-list');
        list.innerHTML = "";
        snapshot.forEach((doc) => {
            const data = doc.data();
            const item = document.createElement('div');
            item.className = 'donation-item';
            item.innerHTML = `
                <div style="width:35px; height:35px; border-radius:50%; background:#222; display:flex; align-items:center; justify-content:center; color:var(--accent-gold); border:1px solid var(--accent-gold); font-weight:700;">${data.name.charAt(0)}</div>
                <div style="flex-grow:1">
                    <div style="display:flex; justify-content:space-between; font-size:12px; font-weight:700;">
                        <span>${data.name}</span>
                        <span style="color:var(--accent-gold)">Rp ${data.amount.toLocaleString()}</span>
                    </div>
                    <div style="font-size:10px; color:#aaa; margin-top:2px;">${data.message || ''}</div>
                </div>`;
            list.appendChild(item);
        });
    });
</script>
